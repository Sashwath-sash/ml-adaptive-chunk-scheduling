  3 /* Deterministic integer work. The result depends on both seed and iteration count. */
  4 uint64_t cpu_work(uint64_t iterations, uint64_t seed) {
  5     uint64_t x = seed ^ UINT64_C(0x9e3779b97f4a7c15);
  6     for (uint64_t i = 0; i < iterations; ++i) {
  7         x ^= x >> 12;
  8         x ^= x << 25;
  9         x ^= x >> 27;
 10         x *= UINT64_C(0x2545f4914f6cdd1d);
 11     }
 12     return x;
 13 }

 15 /* Real independent CPU task: count primes in [start, start + count).
 16    The integer result is deterministic and can be checked across schedulers. */
 17 uint64_t prime_count(uint64_t count, uint64_t start) {
 18     uint64_t total = 0;
 19     for (uint64_t value = start; value < start + count; ++value) {
 20         if (value < 2) continue;
 21         uint64_t prime = 1;
 22         for (uint64_t divisor = 2; divisor <= value / divisor; ++divisor) {
 23             if (value % divisor == 0) { prime = 0; break; }
 24         }
 25         total += prime;
 26     }
 27     return total;
 28 }
